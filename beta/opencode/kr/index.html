// 여러 개의 고성능 AI 모델에 동시에 질의합니다

await workspace.initialize({
    models: [
        "Claude 3.5 Sonnet",
        "GPT-4o",
        "Gemini 1.5 Pro"
    ],
    budgetLimit: "₩6,500/월"
});

// 재귀 알고리즘 최적화 수행
const optimized = await workspace.optimizeCode("./src/algorithm.rs");
console.log("성능 향상:", optimized.performanceGain); // +420% 속도 향상
